## 简介

`pcf8574_io8_async_driver` 是当前仓库中独立发布的 MicroPython 异步包。设备/功能名称和 API 均来自包内运行时文件；它不是对同步 API 的无依据重命名。源包：`input/pcf8574_io8_driver`。

## 主要功能

- 可解析的公开类：`PCF8574`, `PCF8574Async`, `PCF8574IO8`, `SyncAdapterMixin`。
- 可解析的公开调用入口：39 个（详见 API 矩阵）。
- 异步等级：`event_bridge`；依据：IRQ/事件通知由协程消费；总线访问仍可能同步。

## 快速开始

IRQ 只记录待处理状态，不在中断中访问 I2C 或调用用户回调。主循环必须周期性 `await pcf.dispatch_pending_async()`；有待处理事件时在任务中读取端口并调用 `callback(port)`，支持普通函数和协程回调，异常向调用任务传播。多个 IRQ 可以合并为一次当前端口读取；这不是逐边沿事件队列。调度队列满的标记也由该方法消费。结束时 `await pcf.aclose()`。

按 `package.json.urls` 安装后，运行以下示例（与 `code/main.py` 一致）：

```python
"""Preserved source demo follows; runnable async example is appended below."""
"""
try:
    import uasyncio as asyncio
except ImportError:
    import asyncio

async def main():
    # Python env   : MicroPython v1.23.0
    # -*- coding: utf-8 -*-
    # @Time    : 2025/9/23 下午5:44
    # @Author  : 缪贵成
    # @File    : main.py
    # @Description : 8位IO扩展驱动测试文件

    # ======================================== 导入相关模块 =========================================

    # 导入I2C和Pin模块
    from machine import I2C, Pin

    # 导入时间模块

    # 导入PCF8574和PCF8574IO8模块
    from pcf8574 import PCF8574

    # 导入PCF8574IO8模块
    from pcf8574_io8 import PCF8574IO8

    # ======================================== 全局变量 ============================================

    PCF8574_ADDR = None

    # ======================================== 功能函数 ============================================

    # ======================================== 自定义类 ============================================

    # ======================================== 初始化配置 ===========================================

    await asyncio.sleep(3)
    print("FreakStudio:PCF8574 Five-way Button Test Program")

    # 初始化I2C
    i2c = I2C(0, scl=Pin(5), sda=Pin(4), freq=100000)

    # 开始扫描I2C总线上的设备，返回从机地址的列表
    devices_list: list[int] = i2c.scan()
    print("START I2C SCANNER")

    # 若devices list为空，则没有设备连接到I2C总线上
    if len(devices_list) == 0:
        # 若非空，则打印从机设备地址
        print("No i2c device !")
    else:
        # 遍历从机设备地址列表
        print("i2c devices found:", len(devices_list))
    for device in devices_list:
        # 判断设备地址是否为的PCF8574地址
        if 0x20 <= device <= 0x28:
            # 找到的设备是PCF_8574地址
            print("I2c hexadecimal address:", hex(device))
            PCF8574_ADDR = device

    # 初始化PCF8574
    pcf = PCF8574(i2c, PCF8574_ADDR)

    # 检查PCF8574设备是否存在
    try:
        if pcf.check():
            print("PCF8574 detected successfully.")
    except OSError as e:
        print("Error: PCF8574 not found!", e)

    # PCF8574IO8 init
    # 初始化模块引脚状态: PORT0=(0,1), PORT1=(1,0), PORT2=(1,1), PORT3=(0,0)
    ports_init = {0: (1, 1), 1: (1, 1), 2: (1, 1), 3: (1, 1)}
    io8 = PCF8574IO8(pcf, ports_init=ports_init)

    print("PCF8574IO8 initialized with default port states:", io8.ports_state())
    await asyncio.sleep(3)

    # ========================================  主程序  ============================================

    # 控制端口2的引脚2闪烁
    io8.set_port(1, 0)
    await asyncio.sleep(1)
    io8.set_port(1, 2)
    await asyncio.sleep(1)
    io8.set_port(1, 0)
    await asyncio.sleep(1)
    io8.set_port(1, 2)

    while True:
        # 引脚0为低电平时（not 1 → False，not 0 → True）
        if not io8.get_pin(0):
            # 端口2设置为值2
            io8.set_port(1, 2)
            print("Button triggered, set port2 to 1")
        # 引脚0为高电平时
        else:
            # 端口2设置为值0
            io8.set_port(1, 0)
        await asyncio.sleep_ms(20)

if __name__ == '__main__':
    asyncio.run(main())

"""
try:
    import uasyncio as asyncio
except ImportError:
    import asyncio
from machine import I2C, Pin
from pcf8574_async import PCF8574Async
from _gs_pcf8574_io8_pcf8574_io8 import PCF8574IO8


async def main():
    pcf = None
    io8 = None
    try:
        i2c = I2C(0, scl=Pin(5), sda=Pin(4), freq=100000)
        pcf = PCF8574Async(i2c, 0x20)
        if not await pcf.check_async():
            raise RuntimeError("PCF8574 not found")
        io8 = PCF8574IO8(pcf)
        print(io8.ports_state())
        while True:
            await pcf.dispatch_pending_async()
            print(io8.get_pin(0))
            io8.set_port(1, 2)
            await asyncio.sleep_ms(500)
            io8.set_port(1, 0)
            await asyncio.sleep_ms(500)
    finally:
        if io8 is not None:
            io8.deinit()
        if pcf is not None:
            await pcf.aclose()


asyncio.run(main())
```

## 超时、取消、并发与残余阻塞

任何 `await` 不会自动使底层 I2C/SPI/UART/Pin 操作变为非阻塞。取消时应停止后台任务并在 `finally` 释放资源；没有实现超时参数的 API 不应被假定具有超时。协议命令、帧界限、ACK、静默间隔及错误含义只以当前实现和源示例可证明的内容为准。
